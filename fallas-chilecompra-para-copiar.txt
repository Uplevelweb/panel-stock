-- 1) TABLA (una fila por suscriptor y dia con falla de ChileCompra)
create table if not exists public.fallas_dia (
  dia date not null,
  suscriptor_id uuid not null references public.suscriptores(id) on delete cascade,
  motivo text,
  creado_en timestamptz not null default now(),
  aviso_en timestamptz,
  aviso_final_en timestamptz,
  con_correo boolean not null default false,
  resuelto_en timestamptz,
  primary key (dia, suscriptor_id)
);
alter table public.fallas_dia enable row level security;  -- sin politicas: solo service_role
